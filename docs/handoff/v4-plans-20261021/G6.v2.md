def _sha(text):
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def test_v4_one_version_string_names_the_v4_protocol():
    assert VERSION == "fivebench-coevolution-v4" and package.VERSION is VERSION
    assert prelaunch.TEXTS_VERSION == VERSION + "-texts" and prelaunch.IMPORT_VERSION == VERSION + "-texts-import"

    assert kinds == {"manifest", "smoke", "chain", "unit", "evaluation", "test-panel", "cell", "sheet-qualification",

    assert len(expected) == 14 and _sha(json.dumps(expected, sort_keys=True)) == FROZEN["expected_calls"]
    assert (ROLE_MODEL, ROLE_MAX_TOKENS) == ("glm-5.3", 16384)


def test_v4_b15_the_round_bound_binds_only_after_a_round_cut_short():

    assert (RESEARCH_MAX_ROUNDS, RESEARCH_MAX_CALLS, research.CALLS_PER_ROUND) == (3, 8, 3)
    assert RESEARCH_MAX_CALLS // research.CALLS_PER_ROUND == 2 < RESEARCH_MAX_ROUNDS
    assert 1 + 1 + research.CALLS_PER_ROUND <= RESEARCH_MAX_CALLS  # two one-call rounds leave room for a third

    from tests.coevo5_fakes import roles

    model, script = roles(tmp_path, [{"a": 1}, {"a": 2}])

    assert model.ask("tester", "s", {}, label="t1", temperature=ROLE_VOTE_TEMPERATURE) == {"a": 2}
    assert script.temperatures == [0, ROLE_VOTE_TEMPERATURE] and script.temperatures[-1] == 0.7
    assert script.seen[-1] == ("s", {})
