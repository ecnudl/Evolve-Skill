### 1.3 NEW module `skillopt/coevo5/applicability.py` (complete)

```python
"""The applicability judge (registration v4 P1): a rule's trigger items from its condition, not from diagnosis codes.

A v4 rule reads "When <when>: <do>." (optionally "... Otherwise: <otherwise>."), ``when`` being an observable
condition on the task input the solver sees. v1-v3 (§4 step 7) froze a rule's trigger items as the check-half items
with a failed run of a targeted diagnosis code; v4 freezes them as the check-half items -- passed and failed alike --
on which this judge (GLM-5.3, a method role) finds the condition to hold on the item's public input: the bench's
public task card and the task exactly as the examiner's diagnosis sees it (``examiner.task_view`` with the
diagnosis' context bound, ``DIAGNOSE_CONTEXT_CHARS``; ALFWorld: the episode's initial observation from the training
execution's trace), never a reference, an answer or a score. Each item is judged by ``ROLE_VOTES`` independent
samples (P4, ``roles.vote``): reason-then-label JSON, the majority label; no majority is "not_applicable".
``MIN_TRIGGER`` applies to the trigger set (fewer: ``insufficient_support``, the counts recorded). The rule's C2
"rest" panel is frozen here too: the check-half items judged not applicable, minus the rule's own evidence items
(v3 §7). Per rule the record carries, informational only (no gate criterion reads them): ``when_matched`` (the
applicable items), ``when_precision`` (the applicable items that are check-half failures of one of the rule's
codes, over the applicable items) and ``when_recall`` (the same, over all check-half failures of those codes). The
fixed prompt carries one worked example about an imaginary colour-sorting task (P2), hashed with the role's prompt
identity like the rest of it.

Cost (v4 §0): (rules judged) x (check-half items) x ROLE_VOTES calls per step; ``protocol.LEARNING_CAP`` documents
the arithmetic. The judge is asked only for a rule that passed the leakage and length checks (``step.prepare``).
"""
from __future__ import annotations

from fractions import Fraction

from skillopt.continual_eval.core import require

from .cells import run_parallel
from .examiner import task_view
from .protocol import APPLICABILITY_REASON_CHARS, DIAGNOSE_CONTEXT_CHARS, MAX_WHEN_CHARS, MIN_TRIGGER, ROLE_VOTES
from .roles import Invalid

APPLICABLE, NOT_APPLICABLE = "applicable", "not_applicable"
LABELS = (APPLICABLE, NOT_APPLICABLE)
# the record of a rule the judge was not asked about (it may not be tested anyway: a leak or an overlength text)
NOT_JUDGED = {"trigger": [], "rest": [], "applicability": None, "when_matched": None, "when_precision": None,
              "when_recall": None}

JUDGE_SYSTEM = (
    "You judge whether a condition holds for one task. A guidance rule for a language model that solves tasks has "
    "the form \"When <condition>: <action>.\"; you receive the condition alone, the public description of this kind "
    "of task (what the solver is told), and one task exactly as the solver sees it -- never its answer. Decide "
    "whether the condition holds for this task from its input alone: what the task asks, the material it supplies "
    "and how they are written. Do not solve the task, do not guess its answer, and do not judge whether the action "
    "would help. A condition that refers to something the solver cannot see before answering (the correct answer, "
    "whether an answer is right, a score), or that cannot be decided from the input, does not hold. Reply with one "
    "JSON object only, the reason before the label: {\"reason\": a short reason of at most 300 characters, "
    "\"label\": \"applicable\" or \"not_applicable\"}.\n\n"
    "Worked example (an imaginary colour-sorting task, unrelated to any task you will see). Condition: \"the task "
    "lists more than five colours to sort\". Task: {\"instruction\": \"Sort these colours from warm to cold: red, "
    "teal, amber, violet, lime, coral, navy\"}. Reply: {\"reason\": \"The instruction lists seven colours, which is "
    "more than five.\", \"label\": \"applicable\"}.")


def judgment(value):
    """The judge's reply (P4: a reason, then one of ``LABELS``); B5: extra keys are tolerated and a reason over
    ``APPLICABILITY_REASON_CHARS`` is clipped for the archive, never a reason to void."""
    if type(value) is not dict or value.get("label") not in LABELS:
        raise Invalid("label must be \"applicable\" or \"not_applicable\"")
    if type(value.get("reason")) is not str:
        raise Invalid("reason must be a string")
    return {"reason": value["reason"][:APPLICABILITY_REASON_CHARS], "label": value["label"]}


def view(benchmark, public, *, trace=None):
    """The item's public input exactly as the diagnosis sees it (``examiner.diagnose_one``), never a reference."""
    return task_view(benchmark, public, trace=trace, context_chars=DIAGNOSE_CONTEXT_CHARS.get(benchmark))


def views(benchmark, tasks, answers, items):
    """{item: its public view} for ``items``; ``answers`` (``step.answers``, by item then run) lend ALFWorld's
    initial observation (the first run's trace; an item without one is shown an empty observation)."""
    traces = {}
    for answer in answers:
        traces.setdefault(answer["item"], answer.get("trace"))
    return {item: view(benchmark, tasks[item]["public"], trace=traces.get(item)) for item in items}


def judge(roles, card, when, item_view, *, label):
    """One item's applicability: ``roles.vote``'s result {"label", "votes", "tally"} (P4; a tie is not applicable).
    The message is bounded by construction: the view is the diagnosis' clipped view (at most 32,000 characters of
    OfficeQA pages), far under ``ROLE_INPUT_BYTES``."""
    require(type(when) is str and when.strip() and len(when) <= MAX_WHEN_CHARS,
            "A condition of 1-%d characters" % MAX_WHEN_CHARS)
    payload = {"task_card": card, "condition": when, "task": item_view}
    return roles.vote("applicability", JUDGE_SYSTEM, payload, label=label, check=judgment,
                      label_of=lambda value: value["label"], tie=NOT_APPLICABLE)


def derive(roles, card, when, views_, *, check_items, failed_codes, codes, evidence, label, workers):
    """A rule's frozen panels from its condition over every check-half item (``check_items``, sorted; ``views_``:
    {item: view}), with the judge's votes and the condition statistics.

    ``failed_codes``: {check-half failed item: its diagnosis code} (B8: one code per item); ``codes``: the rule's
    codes (B1); ``evidence``: the rule's cited items (left out of the rest panel, v3 §7). Returns {"trigger": the
    applicable items, "rest": the not-applicable items minus the evidence, "judged": {item: the vote result},
    "applicability": the counts, "when_matched", "when_precision", "when_recall"} (the last two exact fractions as
    strings, None when their denominator is 0)."""
    require(list(check_items) == sorted(set(check_items)) and set(check_items) <= set(views_),
            "Sorted distinct check-half items, each with a view")
    found = run_parallel(lambda item: judge(roles, card, when, views_[item], label="%s/%s" % (label, item)),
                         list(check_items), workers=workers)
    judged = dict(zip(check_items, found))
    trigger = [i for i in check_items if judged[i]["label"] == APPLICABLE]
    own = set(evidence)
    rest = [i for i in check_items if judged[i]["label"] != APPLICABLE and i not in own]
    targeted = {i for i, code in failed_codes.items() if code in set(codes)}
    hit = len(targeted & set(trigger))
    ties = sum(1 for r in judged.values() if 2 * max(r["tally"].values(), default=0) <= ROLE_VOTES)
    counts = {"check_items": len(check_items), "applicable": len(trigger),
              "not_applicable": len(check_items) - len(trigger), "tie": ties,
              "evidence_in_check_half": len(own & set(check_items)), "min_trigger": MIN_TRIGGER}
    return {"trigger": trigger, "rest": rest, "judged": judged, "applicability": counts,
            "when_matched": len(trigger),
            "when_precision": str(Fraction(hit, len(trigger))) if trigger else None,
            "when_recall": str(Fraction(hit, len(targeted))) if targeted else None}
```
